\begin{frame}{Clinical Motivation: Beyond "One-Size-Fits-All" Guidelines}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.6em}
        \item \textbf{Global Burden:} Over 537 million adults live with diabetes; postprandial glycemic volatility drives micro- and macro-vascular complications.
        \item \textbf{High Inter-Individual Variability:} Studies (Zeevi et al., Hall et al.) show identical carbohydrate meals cause dramatically different glucose spikes across patients.
        \item \textbf{Complex Drivers:} Glycemic response depends on multi-modal factors:
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item Insulin resistance (HOMA-IR, HbA1c)
          \item Pre-meal glucose momentum \& physical activity
          \item Meal macronutrient balance (fiber, fat, protein)
          \item Gut microbiome bacterial composition
        \end{itemize}
      \end{itemize}
    \end{column}
    
    \begin{column}{0.42\textwidth}
      \begin{alertblock}{The Clinical Need}
        Static universal diets fail because human metabolic dynamics are highly personalized. Patients need a \textbf{prospective digital twin} to predict post-meal excursions \emph{before} eating.
      \end{alertblock}
      \vspace{0.2cm}
      \begin{exampleblock}{Goal of GlycoTwin}
        Forecast full continuous 120-minute postprandial glucose curves and simulate "What-If" pre-meal lifestyle modifications.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 4: Limitations of Existing Approaches
% -----------------------------------------------------------------------------
\begin{frame}{Limitations of Existing Continuous Glucose Monitoring (CGM)}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Reactive Rather than Proactive Alerts}
        \begin{itemize}
          \item Standard CGMs (Dexcom, Libre) trigger threshold alarms \textbf{only after} hyperglycemia ($>180$ mg/dL) has occurred.
          \item Food has already been absorbed; damage cannot be prevented retrospectively.
        \end{itemize}
      \end{block}
      \vspace{0.1cm}
      \begin{block}{2. Isolated Glucose Tracking}
        \begin{itemize}
          \item Glucose curves are analyzed in isolation without integrating meal nutrients, smartwatch fitness streams, and blood labs.
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{3. Inability to Model Gut Microbiota}
        \begin{itemize}
          \item Gut sequencing yields presence/absence for 1,900+ bacterial taxa in small clinical cohorts ($N < 100$).
          \item Traditional linear PCA fails on sparse binary data and misses non-linear metabolic microbial interactions.
        \end{itemize}
      \end{block}
      \vspace{0.1cm}
      \begin{alertblock}{Baseline Offset Bias in Deep Models}
        Training standard networks on raw absolute glucose leads to average mean collapse ($\approx 120$ mg/dL), flattening true postprandial peaks.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Problem Statement \& Key Research Objectives}
  \begin{block}{Problem Statement}
    To engineer a personalized, multi-modal digital twin framework that models multi-source physiological and metagenomic data to predict continuous postprandial glucose trajectories and simulate proactive lifestyle interventions before meal consumption.
  \end{block}
  
  \vspace{0.15cm}
  \textbf{Key Research Objectives:}
  \begin{enumerate}\setlength{\itemsep}{0.4em}
    \item \textbf{Multi-Modal Data Pipeline:} Synchronize dual CGM streams (1-min resolution), wearable activity, meal logs, blood labs, and 1,979 gut microbial taxa.
    \item \textbf{Microbiome Representation Learning:} Build an unsupervised $\beta$-VAE to compress 1,979 binary bacterial species into 8 continuous, meaningful latent dimensions.
    \item \textbf{TransformerForecaster Architecture:} Formulate a hybrid self-attention network using relative delta targets ($\Delta G_t = G_t - G_0$) and composite excursion loss.
    \item \textbf{Rigorous Benchmark Validation:} Evaluate strictly out-of-sample on unseen human subjects under 5-Fold Group K-Fold cross-validation and Clarke Grid safety.
    \item \textbf{Causal "What-If" Simulation \& Mobile App:} Enable pre-meal lifestyle counterfactual exploration through a lightweight, mobile-first Progressive Web App.
  \end{enumerate}
\end{frame}

\begin{frame}{Unsupervised Gut Microbiome Latent Modeling ($\beta$-VAE)}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.35em}
        \item \textbf{The Data Challenge:} 1,979 sparse binary bacterial taxa across 45 participants ($p \gg n$ curse of dimensionality).
        \item \textbf{Model Formulation:} Deep $\beta$-Variational Autoencoder compressing $1,979 \to 8$ continuous latent variables ($\mathbf{z} \in \mathbb{R}^8$).
        \item \textbf{Denoising Taxa-Masking:} Stochastic input zero-masking ($p=0.15$) compels the network to learn co-occurring bacterial consortia.
        \item \textbf{Cyclic $\beta$-Annealing:} Controls KL divergence penalty:
        $$\mathcal{L}_{\text{VAE}} = \mathcal{L}_{\text{BCE}}(\mathbf{x}, \hat{\mathbf{x}}) + \beta \cdot D_{\text{KL}}\big(q(\mathbf{z}|\mathbf{x}) \,\|\, p(\mathbf{z})\big)$$
        \item \textbf{Performance:} Perfect out-of-sample reconstruction: \textbf{ROC-AUC: 1.0000}, \textbf{PR-AUC: 1.0000}.
      \end{itemize}
    \end{column}

\begin{frame}{TransformerForecaster: Dynamic \& Static Modality Fusion}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \begin{block}{Dynamic Temporal Attention Encoder}
        \begin{itemize}\setlength{\itemsep}{0.25em}
          \item Pre-meal telemetry: $\mathbf{X}_{\text{seq}} \in \mathbb{R}^{60 \times 3}$ (Glucose, Heart Rate, METs).
          \item Linear projection to $d_{\text{model}} = 64$ with sinusoidal positional encodings.
          \item 2-Layer Multi-Head Self-Attention ($h=4$ heads) + GELU Feed-Forward Network ($d_{\text{ff}} = 128$).
          \item Temporal Context Pooling: $\mathbf{h}_{\text{temp}} = \frac{1}{60}\sum \mathbf{E}_{L,t} \in \mathbb{R}^{64}$.
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{Gated Residual Network (GRN) for Static Data}
        \begin{itemize}\setlength{\itemsep}{0.25em}
          \item Static input: $\mathbf{x}_{\text{stat}} \in \mathbb{R}^{61}$ (Nutrients, Labs, VAE Latents).
          \item Suppresses noisy features while amplifying high-impact biomarkers (Carbs, HOMA-IR):
          $$\mathbf{g} = \sigma(\mathbf{W}_g \mathbf{\eta}_2 + \mathbf{b}_g) \quad \text{(Gating Mechanism)}$$
          $$\mathbf{h}_{\text{stat}} = \text{LayerNorm}(\mathbf{W}_{\text{skip}}\mathbf{x}_{\text{stat}} + \mathbf{g} \odot \mathbf{\eta}_2)$$
        \end{itemize}
      \end{block}
      \vspace{0.1cm}
      \begin{exampleblock}{Fusion \& Multi-Horizon Head}
        $\mathbf{h}_{\text{fused}} = [\mathbf{h}_{\text{temp}} \,\|\, \mathbf{h}_{\text{stat}}] \in \mathbb{R}^{128} \longrightarrow \hat{\mathbf{\Delta G}} \in \mathbb{R}^{24}$
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 10: Eliminating Baseline Bias & Composite Loss
% -----------------------------------------------------------------------------
\begin{frame}{Eliminating Baseline Bias \& Composite Excursion Loss}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Problem: Baseline Offset Bias}
        Baseline fasting glucose varies widely ($85$ mg/dL healthy vs. $165$ mg/dL diabetic). Models trained on absolute glucose collapse to population average ($\approx 120$ mg/dL).
      \end{alertblock}
      
      \begin{exampleblock}{Solution: Residual Delta Target}
        Train network strictly on relative excursion rise:
        $$\Delta G(t) = G(t) - G_0, \quad \forall t \in \{5, \dots, 120\text{m}\}$$
        Reconstruct during inference without drift:
        $$\hat{G}(t) = G_0 + \max\big(0, \hat{\Delta G}(t)\big)$$
      \end{exampleblock}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{Composite Excursion Loss Function}
        Standard MSE flattens acute spikes. Our loss enforces both numerical and physiological fidelity:
        $$\mathcal{L}_{\text{total}} = \mathcal{L}_{\text{MSE}} + \lambda_1 \mathcal{L}_{\text{peak}} + \lambda_2 \mathcal{L}_{\text{deriv}}$$
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item $\mathcal{L}_{\text{peak}} = \big| \max(\hat{\mathbf{Y}}) - \max(\mathbf{Y}) \big|$ (Peak Height)
          \item $\mathcal{L}_{\text{deriv}} = \frac{1}{H-1}\sum | \Delta\hat{Y}_k - \Delta Y_k |^2$ (Velocity)
          \item Weights: $\lambda_1 = 0.6$, $\lambda_2 = 0.3$.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 11: Experimental Protocol & Baselines

\begin{frame}{Experimental Setup: 5-Fold Group K-Fold Protocol}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \begin{block}{Strictly Unseen Human Validation}
        \begin{itemize}\setlength{\itemsep}{0.3em}
          \item Random splitting leads to data leakage because models memorize patient-specific baseline glucose.
          \item We enforced \textbf{5-Fold Group K-Fold Cross-Validation strictly grouped by Subject ID}:
          \begin{itemize}
            \item Train: $\sim 36$ participants ($\approx 1,306$ meal events)
            \item Test: 9 completely unseen participants ($\approx 327$ meals)
            \item Zero subject overlap between train \& test sets.
          \end{itemize}
          \item Metrics aggregated over all 1,633 meal events.
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{Benchmarked Baseline Architectures}
        \begin{enumerate}\setlength{\itemsep}{0.3em}
          \item \textbf{MLP Baseline:} 3-layer dense network with summary dynamic features.
          \item \textbf{LSTM Forecaster:} 2-layer recurrent LSTM ($h=64$) + static dense head.
          \item \textbf{GRU Forecaster:} 2-layer GRU ($h=64$) with matched recurrent parameters.
          \item \textbf{Transformer (PCA):} Our transformer using linear 8-PCA microbiome features.
          \item \textbf{Transformer ($\beta$-VAE --- Proposed):} Full architecture with non-linear gut manifold.
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

      \begin{block}{Structural Causal Models (Pearl's $do$-Calculus)}
        Clones current patient state $\mathbf{x}_0 = (\mathbf{X}_{\text{seq}}, \mathbf{x}_{\text{stat}}, G_0)$ and evaluates interventional distributions:
        $$P\big(\mathbf{Y} \,\big|\, do(\text{Meal}=\mathbf{m}^*), do(\text{Activity}=\mathbf{a}^*), \mathbf{x}=\mathbf{x}_0\big)$$
      \end{block}
      
      \textbf{The 4 Pre-Meal Counterfactual Scenarios:}
      \begin{enumerate}\setlength{\itemsep}{0.2em}
        \item \textbf{Scenario 1: Planned Meal (Control)}
        \item \textbf{Scenario 2: -50\% Carbohydrate Portion Cut}
        \item \textbf{Scenario 3: +30-Minute Post-Meal Walk (3.5 METs)}
        \item \textbf{Scenario 4: Dual Combined Intervention}
      \end{enumerate}
    \end{column}
    
    \begin{column}{0.45\textwidth}
      \begin{exampleblock}{Clinical Utility Score ($U \in [0, 1]$)}
        Scores and ranks the 4 scenarios for the user:
        $$U = 0.50 \left(\frac{\text{TIR}}{100}\right) + 0.30 \max\left(0, 1 - \frac{G_{\max} - 70}{180}\right) + 0.20 \max\left(0, 1 - \frac{\Delta G_{\max}}{100}\right)$$
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item $\text{TIR}$: \% Time-in-Range ($70-180$ mg/dL)
          \item $G_{\max}$: Absolute peak spike penalty
          \item $\Delta G_{\max}$: Excursion amplitude penalty
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
